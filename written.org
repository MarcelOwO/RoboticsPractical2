*** B1: Properties of Homogeneous Transformations

**** a.

\begin{equation}
\begin{aligned}
{}^{A}\mathbf{r}_{AP} &= ArAB + \mathbf{C}_{AB} BrBP \\[6pt]
ArAP - ArAB &= C_{AB} BrBP \\[6pt]
(ArAP - ArAB) C^{-1}_{AB} &= (C_{AB} BrBP) C^{-1}_{AB} \\[6pt]
(ArAP - ArAB) C^{-1}_{AB} &=  BrBP \\[6pt]
BrBP &= C^{T}_{AB}ArAP - C^{T}_{AB}ArAB
\end{aligned}
\end{equation}

\[
T_{AB} = \begin{bmatrix}
C_{AB} & ArAB \\
0 & 1 \end{bmatrix}
\]

\[
T_{BA} = \\
T^{-1}_{AB} = \\ \begin{bmatrix}
C^{T}_{AB} & -C^{T}_{AB} ArAB \\
0 & 1 \end{bmatrix}
\]

\begin{equation}
\begin{aligned}
T_{AB}T_{BA} &=
\begin{bmatrix} C & r \\ 0 & 1 \end{bmatrix}
\begin{bmatrix} C^{T} & -C^{T}r \\ 0 & 1 \end{bmatrix} \\
&= \begin{bmatrix} CC^{T} & -CC^{T}r+r \\ 0 & 1 \end{bmatrix}
\end{aligned}
\end{equation}

\[
CC^{T} = I_{3}
\]

\[
= \begin{bmatrix} I_{3} & -r +r \\ 0 & 1 \end{bmatrix} = I_{4}
\]

import numpy as np
from scipy.spatial.transform import Rotation

def inverse_transform(T_AB):
    C = T_AB[:3,:3]
    r = T_AB[:3,3]

    T_BA = np.eye(4)
    T_BA[:3,:3] = C.T
    T_BA[:3,3] = -C.T @ r
    return T_BA

A = np.array([0,0,0])
B = np.array([1,0,0])

ArAB = B-A

T_AB[:3,3] = ArAB
